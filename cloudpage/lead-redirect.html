%%[ /* ===================================================================== CloudPage : génération du lead au clic puis redirection vers le funnel Flux : 1. Le lien de l'email (CloudPagesURL) porte une clé opaque chiffrée (?qs=...) 2. GET -> page de confirmation avec bouton (évite les leads créés par les scanners de liens : Safe Links, Proofpoint, ...) 3. POST -> go=1 : recherche d'un lead valide (log DE, puis Service Cloud), création si absent, redirection vers le funnel avec le token Anti-bot supplémentaire : le lien porte l'heure d'envoi (paramètre t, GetSendTime()). Un POST reçu moins de @cfgMinSeconds secondes après l'envoi est ignoré (aucun lead) et tracé dans Lead_Error_Log. Aucune donnée personnelle ne transite par l'URL. Doit être publiée sur un domaine SFMC (Marketing Cloud Connect actif dans la BU). ===================================================================== */ VAR @cfgFunnelUrl, @cfgGenericParams, @cfgMaxAgeDays, @cfgTokenField, @cfgServerUtcOffset VAR @cfgDeContacts, @cfgDeLog, @cfgDeErrors, @cfgLeadSource VAR @state, @cle, @rows, @row, @cid, @prog, @dateFin, @go VAR @token, @leadId, @url, @fromCache, @created, @errorMsg VAR @logRows, @logRow, @logDate VAR @cutoffD, @cutoff, @leads, @leadRow, @tr VAR @nom, @prenom, @email, @tel, @societe VAR @sendTs, @nowTs, @secsSinceSend, @cfgMinSeconds /* --------------------------------------------------------------------- Configuration (à adapter par environnement) --------------------------------------------------------------------- */ SET @cfgFunnelUrl = "https://funnel.exemple.be/souscription" /* URL du funnel */ SET @cfgGenericParams = "partner=hesitating" /* paramètres du lien générique (lead non obtenu) */ SET @cfgMaxAgeDays = 60 /* marge : les leads SF sont valides 70 j */ SET @cfgTokenField = "Token__c" /* champ du Lead qui porte le token funnel */ SET @cfgServerUtcOffset = 6 /* Now() SFMC = CST (UTC-6, sans DST) */ SET @cfgMinSeconds = 5 /* POST < 5 s après l'envoi = bot ; 0 = désactivé */ SET @cfgLeadSource = "Programme migration" SET @cfgDeContacts = "Programme_Contacts" SET @cfgDeLog = "Lead_Log" SET @cfgDeErrors = "Lead_Error_Log" /* --------------------------------------------------------------------- 1. Validation de la clé et du programme --------------------------------------------------------------------- */ SET @state = "invalid" SET @errorMsg = "" SET @cle = Trim(RequestParameter("k")) IF NOT EMPTY(@cle) AND Length(@cle) == 36 THEN SET @rows = LookupRows(@cfgDeContacts, "CleOpaque", @cle) IF RowCount(@rows) == 1 THEN SET @row = Row(@rows, 1) SET @dateFin = Field(@row, "DateFinProgramme") /* Programme terminé (jour de fin inclus) */ IF NOT EMPTY(@dateFin) AND DateDiff(@dateFin, Now(), "D") > 0 THEN SET @state = "expired" ELSE SET @go = RequestParameter("go") IF @go == "1" THEN SET @state = "process" ELSE SET @state = "confirm" ENDIF ENDIF ENDIF ENDIF /* Clé inconnue ou programme terminé : lien générique du funnel */ IF @state == "invalid" OR @state == "expired" THEN Redirect(Concat(@cfgFunnelUrl, "?", @cfgGenericParams), false) ENDIF /* --------------------------------------------------------------------- 1b. Anti-bot : secondes écoulées entre l'envoi (paramètre t) et maintenant. DateDiff() AMPScript ne descend pas sous la minute : calcul en SSJS. Les deux dates sont au format yyyy-MM-dd HH:mm:ss, heure serveur (CST). Absent ou illisible : -1 (aucun contrôle). Pas de saut de ligne entre les blocs : Redirect() doit précéder toute sortie. --------------------------------------------------------------------- */ SET @secsSinceSend = -1 SET @sendTs = Trim(RequestParameter("t")) SET @nowTs = FormatDate(Now(), "yyyy-MM-dd HH:mm:ss", "", "fr-BE") ]%%%%[ /* POST trop rapide : pas de lead, on repasse sur la page de confirmation */ IF @state == "process" AND @cfgMinSeconds > 0 AND @secsSinceSend >= 0 AND @secsSinceSend < @cfgMinSeconds THEN InsertData(@cfgDeErrors, "CleOpaque", @cle, "ContractId", Field(@row, "ContractId"), "Message", Concat("SUSPECT : POST ", @secsSinceSend, " s après l'envoi"), "DateErreur", Now()) SET @state = "confirm" ENDIF /* --------------------------------------------------------------------- 2. Création / récupération du lead (uniquement après clic sur le bouton) --------------------------------------------------------------------- */ IF @state == "process" THEN SET @cid = Field(@row, "ContractId") SET @prog = Field(@row, "ProgrammeRef") SET @token = "" SET @leadId = "" SET @fromCache = false SET @created = false /* 2a. Cache : DE de log (0 appel API Salesforce) */ SET @logRows = LookupRows(@cfgDeLog, "CleOpaque", @cle) IF RowCount(@logRows) > 0 THEN SET @logRow = Row(@logRows, 1) SET @logDate = Field(@logRow, "DateCreation") IF NOT EMPTY(Field(@logRow, "Token")) AND NOT EMPTY(@logDate) THEN IF DateDiff(@logDate, Now(), "D") < @cfgMaxAgeDays THEN SET @token = Field(@logRow, "Token") SET @leadId = Field(@logRow, "LeadId") SET @fromCache = true ENDIF ENDIF ENDIF /* 2b. Lead existant et encore valide dans Service Cloud */ IF EMPTY(@token) THEN /* CreatedDate est en UTC : on convertit l'heure serveur (CST) */ SET @cutoffD = DateAdd(DateAdd(Now(), @cfgServerUtcOffset, "H"), Multiply(@cfgMaxAgeDays, -1), "D") SET @cutoff = Concat(FormatDate(@cutoffD, "yyyy-MM-dd", "", "fr-BE"), "T", FormatDate(@cutoffD, "HH:mm:ss", "", "fr-BE"), "Z") SET @leads = RetrieveSalesforceObjects("Lead", Concat("Id,", @cfgTokenField), "Contract_Ref__c", "=", @cid, "Programme_Ref__c", "=", @prog, "CreatedDate", ">", @cutoff) IF RowCount(@leads) > 0 THEN SET @leadRow = Row(@leads, 1) SET @leadId = Field(@leadRow, "Id") SET @token = Field(@leadRow, @cfgTokenField) ENDIF ENDIF /* 2c. Création du lead si aucun lead valide */ IF EMPTY(@token) AND EMPTY(@leadId) THEN SET @nom = Field(@row, "Nom") SET @prenom = Field(@row, "Prenom") SET @email = Field(@row, "Email") SET @tel = Field(@row, "Telephone") SET @societe = Field(@row, "Societe") IF EMPTY(@societe) THEN SET @societe = "Particulier" ENDIF /* Company est obligatoire sur Lead */ SET @leadId = CreateSalesforceObject("Lead", 9, "LastName", @nom, "FirstName", @prenom, "Email", @email, "Phone", @tel, "Company", @societe, "LeadSource", @cfgLeadSource, "Contract_Ref__c", @cid, "Programme_Ref__c", @prog, "Description", Concat("Créé au clic - clé ", @cle)) IF EMPTY(@leadId) OR @leadId == 0 THEN SET @leadId = "" SET @errorMsg = "CreateSalesforceObject a échoué" ELSE SET @created = true ENDIF ENDIF /* 2d. Le token est généré côté Salesforce : on le relit si nécessaire */ IF NOT EMPTY(@leadId) AND EMPTY(@token) THEN SET @tr = RetrieveSalesforceObjects("Lead", @cfgTokenField, "Id", "=", @leadId) IF RowCount(@tr) > 0 THEN SET @token = Field(Row(@tr, 1), @cfgTokenField) ENDIF IF EMPTY(@token) THEN SET @errorMsg = "Token vide après création/lecture du lead" ELSEIF @created THEN /* On ne logue que les leads créés ici : leur date de création est connue */ UpsertData(@cfgDeLog, 1, "CleOpaque", @cle, "LeadId", @leadId, "Token", @token, "DateCreation", Now()) ENDIF ENDIF /* 2e. Redirection : funnel pré-rempli, ou repli sans pré-remplissage */ IF NOT EMPTY(@token) THEN SET @url = Concat(@cfgFunnelUrl, "?token=", URLEncode(@token)) ELSE /* Lead non obtenu : lien générique du funnel, sans pré-remplissage */ SET @url = Concat(@cfgFunnelUrl, "?", @cfgGenericParams) InsertData(@cfgDeErrors, "CleOpaque", @cle, "ContractId", @cid, "Message", @errorMsg, "DateErreur", Now()) ENDIF Redirect(@url, false) ENDIF ]%% Votre offre

Cette offre est terminée

Le programme n'est plus ouvert à la souscription. Contactez-nous pour connaître les alternatives disponibles.

%%[ ELSE ]%%

Lien invalide

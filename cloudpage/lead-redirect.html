%%[ /* ===================================================================== CloudPage: create the lead on click, then redirect to the sales funnel Flow: 1. The email link (CloudPagesURL) carries an encrypted opaque key (?qs=...) 2. GET -> confirmation page with a button (prevents leads being created by link scanners: Safe Links, Proofpoint, ...) 3. POST -> go=1: look for a valid lead (log DE, then Service Cloud), create it if missing, redirect to the funnel with the token Additional anti-bot check: the link carries the send time (parameter t, GetSendTime()). A POST received less than @cfgMinSeconds seconds after the send is ignored (no lead created) and logged in Lead_Error_Log. No personal data travels in the URL. Must be published on an SFMC domain (Marketing Cloud Connect enabled in the business unit). ===================================================================== */ VAR @cfgFunnelUrl, @cfgGenericParams, @cfgMaxAgeDays, @cfgTokenField, @cfgServerUtcOffset VAR @cfgDeContacts, @cfgDeLog, @cfgDeErrors, @cfgLeadSource VAR @state, @cle, @rows, @row, @cid, @prog, @dateFin, @go VAR @token, @leadId, @url, @fromCache, @created, @errorMsg VAR @logRows, @logRow, @logDate VAR @cutoffD, @cutoff, @leads, @leadRow, @tr VAR @nom, @prenom, @email, @tel, @societe VAR @sendTs, @nowTs, @secsSinceSend, @cfgMinSeconds /* --------------------------------------------------------------------- Configuration (adapt per environment) --------------------------------------------------------------------- */ SET @cfgFunnelUrl = "https://funnel.exemple.be/souscription" /* funnel URL */ SET @cfgGenericParams = "partner=hesitating" /* query string of the generic link (no lead obtained) */ SET @cfgMaxAgeDays = 60 /* safety margin: SF leads are valid for 70 days */ SET @cfgTokenField = "Token__c" /* Lead field holding the funnel token */ SET @cfgServerUtcOffset = 6 /* SFMC Now() = CST (UTC-6, no DST) */ SET @cfgMinSeconds = 5 /* POST < 5 s after send = bot; 0 = disabled */ SET @cfgLeadSource = "Programme migration" SET @cfgDeContacts = "Programme_Contacts" SET @cfgDeLog = "Lead_Log" SET @cfgDeErrors = "Lead_Error_Log" /* --------------------------------------------------------------------- 1. Validate the key and the program --------------------------------------------------------------------- */ SET @state = "invalid" SET @errorMsg = "" SET @cle = Trim(RequestParameter("k")) IF NOT EMPTY(@cle) AND Length(@cle) == 36 THEN SET @rows = LookupRows(@cfgDeContacts, "CleOpaque", @cle) IF RowCount(@rows) == 1 THEN SET @row = Row(@rows, 1) SET @dateFin = Field(@row, "DateFinProgramme") /* Program ended (end date included) */ IF NOT EMPTY(@dateFin) AND DateDiff(@dateFin, Now(), "D") > 0 THEN SET @state = "expired" ELSE SET @go = RequestParameter("go") IF @go == "1" THEN SET @state = "process" ELSE SET @state = "confirm" ENDIF ENDIF ENDIF ENDIF /* Unknown key or ended program: generic funnel link */ IF @state == "invalid" OR @state == "expired" THEN Redirect(Concat(@cfgFunnelUrl, "?", @cfgGenericParams), false) ENDIF /* --------------------------------------------------------------------- 1b. Anti-bot: seconds elapsed between the send (parameter t) and now. AMPscript DateDiff() cannot go below one minute: computed in SSJS. Both dates use the format yyyy-MM-dd HH:mm:ss, server time (CST). Missing or unreadable: -1 (no check). No line break between the blocks: Redirect() must precede any output. --------------------------------------------------------------------- */ SET @secsSinceSend = -1 SET @sendTs = Trim(RequestParameter("t")) SET @nowTs = FormatDate(Now(), "yyyy-MM-dd HH:mm:ss", "", "fr-BE") ]%%%%[ /* POST too fast: no lead, show the confirmation page again */ IF @state == "process" AND @cfgMinSeconds > 0 AND @secsSinceSend >= 0 AND @secsSinceSend < @cfgMinSeconds THEN InsertData(@cfgDeErrors, "CleOpaque", @cle, "ContractId", Field(@row, "ContractId"), "Message", Concat("SUSPECT: POST ", @secsSinceSend, " s after send"), "DateErreur", Now()) SET @state = "confirm" ENDIF /* --------------------------------------------------------------------- 2. Create / retrieve the lead (only after the button is clicked) --------------------------------------------------------------------- */ IF @state == "process" THEN SET @cid = Field(@row, "ContractId") SET @prog = Field(@row, "ProgrammeRef") SET @token = "" SET @leadId = "" SET @fromCache = false SET @created = false /* 2a. Cache: log DE (0 Salesforce API calls) */ SET @logRows = LookupRows(@cfgDeLog, "CleOpaque", @cle) IF RowCount(@logRows) > 0 THEN SET @logRow = Row(@logRows, 1) SET @logDate = Field(@logRow, "DateCreation") IF NOT EMPTY(Field(@logRow, "Token")) AND NOT EMPTY(@logDate) THEN IF DateDiff(@logDate, Now(), "D") < @cfgMaxAgeDays THEN SET @token = Field(@logRow, "Token") SET @leadId = Field(@logRow, "LeadId") SET @fromCache = true ENDIF ENDIF ENDIF /* 2b. Existing, still valid lead in Service Cloud */ IF EMPTY(@token) THEN /* CreatedDate is UTC: convert the server time (CST) */ SET @cutoffD = DateAdd(DateAdd(Now(), @cfgServerUtcOffset, "H"), Multiply(@cfgMaxAgeDays, -1), "D") SET @cutoff = Concat(FormatDate(@cutoffD, "yyyy-MM-dd", "", "fr-BE"), "T", FormatDate(@cutoffD, "HH:mm:ss", "", "fr-BE"), "Z") SET @leads = RetrieveSalesforceObjects("Lead", Concat("Id,", @cfgTokenField), "Contract_Ref__c", "=", @cid, "Programme_Ref__c", "=", @prog, "CreatedDate", ">", @cutoff) IF RowCount(@leads) > 0 THEN SET @leadRow = Row(@leads, 1) SET @leadId = Field(@leadRow, "Id") SET @token = Field(@leadRow, @cfgTokenField) ENDIF ENDIF /* 2c. Create the lead if no valid lead exists */ IF EMPTY(@token) AND EMPTY(@leadId) THEN SET @nom = Field(@row, "Nom") SET @prenom = Field(@row, "Prenom") SET @email = Field(@row, "Email") SET @tel = Field(@row, "Telephone") SET @societe = Field(@row, "Societe") IF EMPTY(@societe) THEN SET @societe = "Particulier" ENDIF /* Company is required on Lead */ SET @leadId = CreateSalesforceObject("Lead", 9, "LastName", @nom, "FirstName", @prenom, "Email", @email, "Phone", @tel, "Company", @societe, "LeadSource", @cfgLeadSource, "Contract_Ref__c", @cid, "Programme_Ref__c", @prog, "Description", Concat("Créé au clic - clé ", @cle)) IF EMPTY(@leadId) OR @leadId == 0 THEN SET @leadId = "" SET @errorMsg = "CreateSalesforceObject failed" ELSE SET @created = true ENDIF ENDIF /* 2d. The token is generated by Salesforce: read it back if needed */ IF NOT EMPTY(@leadId) AND EMPTY(@token) THEN SET @tr = RetrieveSalesforceObjects("Lead", @cfgTokenField, "Id", "=", @leadId) IF RowCount(@tr) > 0 THEN SET @token = Field(Row(@tr, 1), @cfgTokenField) ENDIF IF EMPTY(@token) THEN SET @errorMsg = "Empty token after lead creation/read" ELSEIF @created THEN /* Only log leads created here: their creation date is known */ UpsertData(@cfgDeLog, 1, "CleOpaque", @cle, "LeadId", @leadId, "Token", @token, "DateCreation", Now()) ENDIF ENDIF /* 2e. Redirect: pre-filled funnel, or fallback without pre-fill */ IF NOT EMPTY(@token) THEN SET @url = Concat(@cfgFunnelUrl, "?token=", URLEncode(@token)) ELSE /* No lead obtained: generic funnel link, no pre-fill */ SET @url = Concat(@cfgFunnelUrl, "?", @cfgGenericParams) InsertData(@cfgDeErrors, "CleOpaque", @cle, "ContractId", @cid, "Message", @errorMsg, "DateErreur", Now()) ENDIF Redirect(@url, false) ENDIF ]%% Votre offre
